(()=>{
const $=id=>document.getElementById(id);
const today=()=>{const d=new Date();return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`};
const nowTime=()=>{const d=new Date();return `${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`};
let db=null,kind='食事',editing=null,records=[];
const status=t=>$('saveStatus').textContent=t;
function openDB(){return new Promise((resolve,reject)=>{
 if(!('indexedDB' in window)){reject(new Error('このブラウザではIndexedDBが使えません'));return}
 const req=indexedDB.open('RikuDietLog',1);
 req.onupgradeneeded=()=>{const d=req.result;if(!d.objectStoreNames.contains('records'))d.createObjectStore('records',{keyPath:'id'})};
 req.onsuccess=()=>{db=req.result;resolve(db)};req.onerror=()=>reject(req.error);
})}
function tx(mode='readonly'){return db.transaction('records',mode).objectStore('records')}
function all(){return new Promise((resolve,reject)=>{const r=tx().getAll();r.onsuccess=()=>resolve(r.result);r.onerror=()=>reject(r.error)})}
function put(item){return new Promise((resolve,reject)=>{const r=tx('readwrite').put(item);r.onsuccess=resolve;r.onerror=()=>reject(r.error)})}
function del(id){return new Promise((resolve,reject)=>{const r=tx('readwrite').delete(id);r.onsuccess=resolve;r.onerror=()=>reject(r.error)})}
function setKind(k){
 kind=k;document.querySelectorAll('.tab').forEach(b=>b.classList.toggle('active',b.dataset.kind===k));
 const food=k==='食事',exercise=k==='運動',weight=k==='体重';
 $('mealWrap').style.display=food?'block':'none';
 $('kcalWrap').style.display=food?'block':'none';$('proteinWrap').style.display=food?'block':'none';
 $('extraWrap').style.display=weight||k==='体調'?'block':'none';
 $('detailLabel').textContent=food?'食べたもの':exercise?'運動内容':weight?'メモ（任意）':'内容・体調';
 $('detail').placeholder=food?'例：フランスパン2枚、ヨーグルト、バナナ':exercise?'例：スクワット20回、リングフィット15分':weight?'例：起床後、トイレ後':'例：体調良好、空腹感ふつう';
 $('extraLabel').textContent=weight?'体重（kg）':'補足（任意）';
 $('extra').placeholder=weight?'例：55.0':'例：空腹感ふつう、よく眠れた';
 if(exercise){$('kcalWrap').style.display='none';$('proteinWrap').style.display='none'}
}
function resetForm(){editing=null;$('formTitle').textContent='記録を追加';$('saveBtn').textContent='この記録を保存';$('editNote').style.display='none';$('detail').value='';$('kcal').value='';$('protein').value='';$('extra').value='';$('time').value=nowTime();setKind(kind)}
function esc(s){return String(s??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function render(){
 const day=$('day').value,items=records.filter(x=>x.date===day).sort((a,b)=>a.time.localeCompare(b.time)||a.created-b.created);
 $('count').textContent=items.length;$('entryCount').textContent=items.length+'件';
 const kcal=items.filter(x=>x.kind==='食事'&&x.kcal!==null).reduce((s,x)=>s+x.kcal,0);
 const prot=items.filter(x=>x.kind==='食事'&&x.protein!==null).reduce((s,x)=>s+x.protein,0);
 $('totalKcal').textContent=items.some(x=>x.kind==='食事'&&x.kcal!==null)?Math.round(kcal):'—';
 $('totalProtein').textContent=items.some(x=>x.kind==='食事'&&x.protein!==null)?prot.toFixed(1)+'g':'—';
 if(!items.length){$('entries').innerHTML='<div class="empty">まだ記録がありません。食べたものから追加してみよう。</div>';return}
 $('entries').innerHTML=items.map(x=>`<div class="entry"><div class="time">${esc(x.time)}</div><div class="entrymain"><div class="entrytitle"><span class="kind">${esc(x.kind==='食事'?x.meal:x.kind)}</span>${esc(x.detail||'(内容なし)')}</div><div class="meta">${x.kind==='食事'?[(x.kcal!==null?x.kcal+' kcal':''),(x.protein!==null?'たんぱく質 '+x.protein+'g':'')].filter(Boolean).join(' · '):''}${x.extra?' · '+esc(x.extra):''}</div></div><div class="actions"><button class="secondary" data-edit="${esc(x.id)}">編集</button><button class="secondary danger" data-del="${esc(x.id)}">削除</button></div></div>`).join('');
}
async function refresh(){records=await all();render()}
$('tabs').addEventListener('click',e=>{const b=e.target.closest('[data-kind]');if(b){if(editing)resetForm();setKind(b.dataset.kind)}});
$('day').value=today();$('time').value=nowTime();$('todayBtn').onclick=()=>{$('day').value=today();render()};
$('day').onchange=render;
$('saveBtn').onclick=async()=>{
 const detail=$('detail').value.trim();if(!detail&&kind!=='体重'){alert('内容を一言だけ入力してね。');return}
 const parse=id=>$(id).value.trim()===''?null:Number($(id).value);
 const old=editing?records.find(x=>x.id===editing):null;
 const item={id:editing||crypto.randomUUID(),date:$('day').value,time:$('time').value||nowTime(),kind,meal:kind==='食事'?$('meal').value:null,detail,kcal:kind==='食事'?parse('kcal'):null,protein:kind==='食事'?parse('protein'):null,extra:$('extra').value.trim(),created:old?.created||Date.now(),updated:Date.now()};
 try{await put(item);status('保存済み');await refresh();resetForm()}catch(e){status('保存できません');alert('保存に失敗しました。JSONバックアップを確認してください。')}
};
$('entries').addEventListener('click',async e=>{
 const ed=e.target.closest('[data-edit]'),de=e.target.closest('[data-del]');
 if(ed){const x=records.find(r=>r.id===ed.dataset.edit);if(!x)return;editing=x.id;setKind(x.kind);$('formTitle').textContent='記録を編集';$('saveBtn').textContent='変更を保存';$('editNote').style.display='block';$('time').value=x.time;$('detail').value=x.detail;$('meal').value=x.meal||'朝食';$('kcal').value=x.kcal??'';$('protein').value=x.protein??'';$('extra').value=x.extra||'';window.scrollTo({top:0,behavior:'smooth'})}
 if(de){if(confirm('この記録を削除する？')){await del(de.dataset.del);await refresh();status('保存済み')}}
});
$('exportBtn').onclick=async()=>{const data={app:'RikuDietLog',version:1,exportedAt:new Date().toISOString(),records:await all()};const blob=new Blob([JSON.stringify(data,null,2)],{type:'application/json'});const url=URL.createObjectURL(blob);const a=document.createElement('a');a.href=url;a.download=`diet-log-${today()}.json`;a.click();URL.revokeObjectURL(url)};
$('importBtn').onclick=()=>$('fileInput').click();
$('fileInput').onchange=async e=>{const f=e.target.files[0];if(!f)return;try{const data=JSON.parse(await f.text());if(!Array.isArray(data.records))throw new Error('形式が違います');for(const r of data.records){if(r.id&&r.date)await put(r)}await refresh();status('読み込み完了');alert('記録を読み込みました。既存の同じIDの記録は更新されます。')}catch(err){alert('読み込めませんでした。正しいバックアップJSONか確認してね。')}e.target.value=''};
(async()=>{try{await openDB();await refresh();status('端末内に保存')}catch(e){status('保存機能を利用できません');alert('このブラウザでは保存機能を利用できない可能性があります。Safariなど別のブラウザで開いて試してください。')}})();
})();
